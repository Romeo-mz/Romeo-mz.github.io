function getAllTags(posts) {
  const tagSet = new Set();
  posts.forEach(post => {
    if (post.tags && Array.isArray(post.tags)) {
      post.tags.forEach(tag => tagSet.add(tag));
    }
  });
  return Array.from(tagSet).sort();
}
```

3. Build tag filter buttons dynamically (call inside loadBlogArticles after getAllTags):

function setupTagFilters(posts) {
  const allTags = getAllTags(posts);
  const container = document.getElementById('tagFilters');
  container.innerHTML = '';
  
  allTags.forEach(tag => {
    const button = document.createElement('button');
    button.textContent = tag;
    button.className = 'tag-filter-btn';
    button.dataset.tag = tag;
    button.setAttribute('aria-pressed', 'false');
    
    button.addEventListener('click', () => {
      toggleTag(tag, button);
      applyFiltersAndSort();
    });
    
    container.appendChild(button);
  });
}
```

4. Implement tag toggle logic:
```javascript
function toggleTag(tag, button) {
  const index = state.activeTags.indexOf(tag);
  if (index > -1) {
    state.activeTags.splice(index, 1);
    button.classList.remove('active');
    button.setAttribute('aria-pressed', 'false');
  } else {
    state.activeTags.push(tag);
    button.classList.add('active');
    button.setAttribute('aria-pressed', 'true');
  }
}
```

5. Implement combined filter and sort pipeline:
```javascript
function applyFiltersAndSort() {
  let filtered = [...allPosts]; // Start with all posts
  
  // If search exists and has value, apply search first
  const searchInput = document.getElementById('blogSearch');
  if (searchInput && searchInput.value.trim().length >= 2) {
    const results = fuse.search(searchInput.value.trim());
    filtered = results.map(r => r.item);
  }
  
  // Apply tag filter (OR logic: match ANY selected tag)
  if (state.activeTags.length > 0) {
    filtered = filtered.filter(post => 
      state.activeTags.some(tag => post.tags.includes(tag))
    );
  }
  
  // Apply sorting
  const [sortBy, sortOrder] = state.sortBy.split('-');
  filtered.sort((a, b) => {
    let comparison = 0;
    if (sortBy === 'date') {
      comparison = new Date(b.date) - new Date(a.date);
    } else if (sortBy === 'readTime') {
      const aTime = parseInt(/(\d+)/.exec(a.readTime)?.[1] || '0');
      const bTime = parseInt(/(\d+)/.exec(b.readTime)?.[1] || '0');
      comparison = bTime - aTime;
    }
    return sortOrder === 'desc' ? comparison : -comparison;
  });
  
  renderPosts(filtered);
}
```
